SYSTEM_PROMPT = """
You are classifying posts from a soccer (football) community.
Assign each post to exactly one of the following categories.

analysis: In-depth discussion, statistics, tactical breakdowns, or news reports supported by evidence.
Example: "Looking at the heatmaps, the fullbacks are pushing so high that the center backs are being exposed on every counter-attack."

hot_take: Provocative, controversial, or highly opinionated statements intended to spark debate, often lacking deep evidence.
Example: "Mbappe is overrated and won't win a Ballon d'Or until he leaves for a more competitive league."

reaction: Emotional outbursts, short comments on live events, memes, or brief expressions of excitement/frustration.
Example: "GOOOOAL! WHAT A STRIKE! I can't believe he scored that from 30 yards out!!"

Respond with ONLY the label name.
Do not explain your reasoning.

Valid labels:
analysis
hot_take
reaction
"""

print("Prompt length:", len(SYSTEM_PROMPT), "characters")

def classify_with_groq(text):
    """Classify a single post. Returns a label string or None if unparseable."""
    try:
        response = client.chat.completions.create(
            model="openai/gpt-oss-120b",
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": f"Classify this post:\n\n{text}"},
            ],
            temperature=0,
            max_tokens=1024,
            reasoning_effort="low",
        )
        # Clean the response: lowercase and remove common punctuation/markdown
        raw = response.choices[0].message.content.strip().lower().replace("*", "").replace(".", "")
